import json,sys
r=json.load(open(sys.argv[1],encoding="utf-8"))
idx=[int(x) for x in sys.argv[2].split(",")]
print({k:r.get(k) for k in ("overall","overall_cause","failure_causes","native_evidence","license_mode")})
for s in r["steps"]:
    if s["index"] in idx:
        c=dict(s.get("checks") or {})
        print("=====", s["index"], s["id"], s["verdict"], "reason:", s.get("reason"), "cause:", s.get("cause"), s.get("started_at"), s.get("finished_at"))
        for k,v in c.items():
            print("  ", k, ":", json.dumps(v, ensure_ascii=False)[:int(sys.argv[3]) if len(sys.argv)>3 else 900])
