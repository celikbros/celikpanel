#!/bin/bash
C=pair-accept__bind-arch__pdns-debian13__p5t2
H=$(printf %s "$C" | sha256sum | cut -c1-24)
KH=/var/tmp/cp-pair5-work/cells/$H/ssh-known-hosts
O=/root/cp-pair5/obs/p5t2
sshn() { port=$1; shift; timeout 120 ssh -o BatchMode=yes -o ConnectTimeout=5 -o StrictHostKeyChecking=yes -o UserKnownHostsFile=$KH -i /var/tmp/cp-pair5-key/id_ed25519 -p $port celik@127.0.0.1 "$@"; }
sshn 2201 'sudo -n journalctl --no-pager -o short-iso --since "2026-09-30 04:41:25" --until "2026-09-30 04:42:30" | grep -v -E "audit|kernel: audit" | grep -v -i -E "secret|tsig" | grep -v -E "celik : |for celik from|user celik\b|celik\(uid=1000\)"' > $O/debian13-inspector-journal-zone-delete.txt 2>&1
sshn 2202 'sudo -n journalctl --no-pager -o short-iso --since "2026-09-30 04:41:25" --until "2026-09-30 04:42:30" _COMM=named' > $O/arch-named-journal-zone-delete.txt 2>&1
wc -l $O/debian13-inspector-journal-zone-delete.txt $O/arch-named-journal-zone-delete.txt
