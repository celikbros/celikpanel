#!/bin/bash
# usage: t2coll.sh SINCE   (UTC "YYYY-MM-DD HH:MM:SS", shortly before zone-delete)
SINCE=$1
W=/mnt/c/Users/alice/AppData/Local/Temp/claude/c--CELIKBROS-PROJECTS-celikpanel/ab34f56e-94b5-4834-a9ea-4e8dbe057e7f/scratchpad/pair5
C=pair-accept__bind-arch__pdns-debian13__p5t2
L=p5t2
H=$(printf %s "$C" | sha256sum | cut -c1-24)
KH=/var/tmp/cp-pair5-work/cells/$H/ssh-known-hosts
O=/root/cp-pair5/obs/$L; mkdir -p $O
sshn() { port=$1; shift; timeout 120 ssh -o BatchMode=yes -o ConnectTimeout=5 -o StrictHostKeyChecking=yes -o UserKnownHostsFile=$KH -i /var/tmp/cp-pair5-key/id_ed25519 -p $port celik@127.0.0.1 "$@"; }
bash $W/ro-collect.sh $C $L arch 2202 bind primary > /dev/null
bash $W/ro-collect.sh $C $L debian13 2201 pdns secondary > /dev/null
bash $W/coll-extra.sh $C $L arch 2202 primary bind > /dev/null
bash $W/coll-extra.sh $C $L debian13 2201 secondary pdns > /dev/null
bash $W/diag-options.sh $C $L arch 2202 > /dev/null
bash $W/ledgerjob.sh $C $L arch 2202 > /dev/null
sshn 2202 "sudo -n journalctl -u celikpanel-agent.service --no-pager -o short-iso --since '$SINCE' | grep -v -i secret | tail -150" > $O/arch-agent-journal-zone-delete.txt 2>&1
sshn 2202 "sudo -n journalctl -u celikpanel-panel.service --no-pager -o short-iso --since '$SINCE' | tail -80" > $O/arch-panel-journal-zone-delete.txt 2>&1
sshn 2201 "sudo -n journalctl --no-pager -o short-iso --since '$SINCE' | grep -i -E 'inspect|celikpeer|sudo|sshd|pdns|denied|permission' | grep -v -i -E 'secret|tsig' | tail -150" > $O/debian13-inspector-journal-zone-delete.txt 2>&1
sshn 2201 'bash -s' > $O/debian13-pdns-journal-summary.txt 2>&1 <<'R'
echo "## $(date -u +%FT%TZ) pdns.service journal: line count, loglevel settings (names/values only), last 30 lines"
sudo -n journalctl -u pdns.service --no-pager -o short-iso | wc -l
sudo -n grep -r -h -E '^\s*(loglevel|log-dns-details|log-dns-queries)\s*=' /etc/powerdns 2>&1
sudo -n journalctl -u pdns.service --no-pager -o short-iso | grep -v -i -E 'secret|tsig' | tail -30
R
sshn 2202 'bash -s' > $O/arch-mail-root-readonly.txt 2>&1 <<'R'
echo "## $(date -u +%FT%TZ) mail root path components (metadata only)"
ls -ld --time-style=full-iso /var/mail /var/spool/mail /var/mail/vhosts /var/spool/mail/vhosts 2>&1
readlink -v /var/mail 2>&1
pacman -Qo /var/mail 2>&1
R
bash $W/counts.sh $L > $O/notify-counts.txt 2>&1
ls -la $O
