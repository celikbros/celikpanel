#!/usr/bin/env python3
"""Independent secret scan for pair2 evidence (not the driver's redactor).

usage: scan.py DIR [--expect-planted]
Exit 0 = no finding; 1 = findings (printed as path:line:rule, value masked).
"""
import json, os, re, sys

RULES = [
    ("admin-password-shape", re.compile(r"Cp7(?:!|%21)[A-Za-z0-9_-]{20,}")),
    ("session-cookie-value", re.compile(r"celikpanel_session(?:=|%3D)(?!\[REDACTED\]|%5BREDACTED%5D)[^;\s\"'\,]{4,}")),
    ("set-cookie-header", re.compile(r"\"(?:set-cookie|cookie)\"\s*:\s*\"(?!\[REDACTED\])[^\"]{4,}\"", re.I)),
    ("set-cookie-list", re.compile(r"\"(?:set-cookie|cookie)\"\s*:\s*\[\s*\"(?!\[REDACTED\])[^\"]{4,}\"", re.I)),
    ("auth-scheme-credential", re.compile(r"\b(?:Bearer|Basic)\s+(?!\[REDACTED\])[A-Za-z0-9._~+/=-]{8,}")),
    ("pem-private-key", re.compile(r"-----BEGIN [A-Z0-9 ]*PRIVATE KEY-----")),
    ("openssh-key-blob", re.compile("b3BlbnNzaC1rZXkt" + "djE")),
    ("license-key-shape", re.compile(r"CPK-[A-Za-z0-9]{8,}")),
    ("bind-tsig-secret", re.compile(r"secret\s+\"[A-Za-z0-9+/=]{16,}\"")),
    ("json-secret-key-with-value", re.compile(
        r"\"([A-Za-z0-9_-]*(?:password|passwd|passphrase|secret|token|cookie|private_key|license_key|api_key|apikey|credential|csrf|totp|recovery_code)[A-Za-z0-9_-]*)\"\s*:\s*\"(?!\[REDACTED\])([^\"]{1,})\"",
        re.I)),
]
# Header lists ["Name", "value"] spanning lines (the driver's capture format).
HEADER_PAIR_RE = re.compile(r'\[\s*"(?:set-cookie|cookie|authorization|proxy-authorization|x-csrf-token|x-celikpanel-csrf)"\s*,\s*"(?!\[REDACTED\])[^"]+"\s*\]', re.I)
PUBLIC_SUFFIXES = ("_state", "_status", "_reason", "_required", "_id", "_sha256", "_count", "_at", "_enabled",
                   "_configured", "_present", "_kind", "_type", "_file", "_path", "_mode")
PUBLIC_KEYS = {"public_key", "credential_id", "token_type", "public_key_sha256", "host_key_sha256"}


def scan(root):
    findings = []
    files = 0
    for base, _, names in os.walk(root):
        for name in names:
            path = os.path.join(base, name)
            files += 1
            try:
                data = open(path, "rb").read().decode("utf-8", "replace")
            except OSError as exc:
                findings.append((path, 0, "unreadable", str(exc)))
                continue
            for m in HEADER_PAIR_RE.finditer(data):
                lineno = data.count(chr(10), 0, m.start()) + 1
                findings.append((os.path.relpath(path, root), lineno, "header-pair-secret",
                                 " ".join(m.group(0)[:30].split()) + "..."))
            for lineno, line in enumerate(data.splitlines(), 1):
                for rule, rx in RULES:
                    for m in rx.finditer(line):
                        if rule == "json-secret-key-with-value":
                            key = m.group(1).lower()
                            if key in PUBLIC_KEYS or key.endswith(PUBLIC_SUFFIXES):
                                continue
                        text = m.group(0)
                        masked = text[:18] + "..." if len(text) > 18 else text
                        findings.append((os.path.relpath(path, root), lineno, rule, masked))
    return files, findings


def main():
    root = sys.argv[1]
    files, findings = scan(root)
    for f in findings:
        print("%s:%d:%s:%s" % f)
    print(json.dumps({"root": root, "files_scanned": files, "findings": len(findings),
                      "rules": [r for r, _ in RULES]}))
    return 1 if findings else 0


if __name__ == "__main__":
    sys.exit(main())
