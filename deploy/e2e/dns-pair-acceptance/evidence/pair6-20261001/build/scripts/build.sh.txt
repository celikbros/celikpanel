#!/bin/bash
set -uo pipefail
B=/root/cp-pair6
PC=e50fe50a07746ee6f9cd7dea80b95ccd8999256c
DC=378647894a060e687aff6e0b2dfbfd280ec03924
date -u +%FT%TZ
CELIKPANEL_REPO=$B/repo bash $B/scripts-run/build-dist.sh --acceptance-license $PC > $B/logs/build-dist.log 2>&1
echo "build-dist exit=$?"
date -u +%FT%TZ
tail -30 $B/logs/build-dist.log
D=/var/tmp/cp-pair6-dist/$PC-acceptance-license
ls -la $D
cat $D/guard-refusal.txt
LC_ALL=C sort -k2 $D/web-dist.sha256 | cmp - $B/logs/web-dist-host-sorted.sha256 && echo WEB-DIST-SAME
for kind in product pure; do
  T=$B/test-$kind
  rm -rf $T; mkdir -p $T
  cp -a $B/driver/. $T/
  if [[ $kind == product ]]; then c=$PC; else c=$DC; fi
  git -C $B/repo archive $c web/src | tar -x -C $T
  ( cd $T/deploy/e2e/dns-pair-acceptance && bash scripts/offline-tests.sh ) > $B/logs/offline-tests-$kind.log 2>&1
  echo "offline $kind exit=$?"
  tail -5 $B/logs/offline-tests-$kind.log
done
date -u +%FT%TZ
