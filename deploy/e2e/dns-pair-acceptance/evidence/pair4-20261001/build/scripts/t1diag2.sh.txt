#!/bin/bash
CELL=pair-accept__bind-debian13__bind-arch__p4t1
H=$(printf %s "$CELL" | sha256sum | cut -c1-24)
KH=/var/tmp/cp-pair4-work/cells/$H/ssh-known-hosts
O=/root/cp-pair4/obs/p4t1; mkdir -p $O
sshn() { port=$1; shift; timeout 120 ssh -o BatchMode=yes -o ConnectTimeout=5 -o StrictHostKeyChecking=yes -o UserKnownHostsFile=$KH -i /var/tmp/cp-pair4-key/id_ed25519 -p $port celik@127.0.0.1 "$@"; }
sshn 2201 'sudo -n journalctl -u celikpanel-agent.service --no-pager -o short-iso --since "2026-09-30 02:14:50" | grep -v -i "secret" | tail -80' > $O/debian13-agent-journal-zone-delete.txt 2>&1
sshn 2201 'sudo -n journalctl -u celikpanel-panel.service --no-pager -o short-iso --since "2026-09-30 02:14:50" | tail -60' > $O/debian13-panel-journal-zone-delete.txt 2>&1
sshn 2202 'sudo -n journalctl --no-pager -o short-iso --since "2026-09-30 02:14:50" _COMM=sshd + _COMM=sshd-session + SYSLOG_IDENTIFIER=sshd + SYSLOG_IDENTIFIER=sshd-session | tail -80' > $O/arch-sshd-journal-zone-delete.txt 2>&1
sshn 2202 'sudo -n journalctl --no-pager -o short-iso --since "2026-09-30 02:14:50" | grep -i -E "inspect|celikpanel-peer|rndc|forced|denied|permission" | grep -v -i "secret \"" | tail -60' > $O/arch-inspector-journal-zone-delete.txt 2>&1
wc -l $O/*zone-delete.txt
cat $O/debian13-agent-journal-zone-delete.txt | cut -c1-600
echo =====; cat $O/arch-sshd-journal-zone-delete.txt | cut -c1-400
echo =====; cat $O/arch-inspector-journal-zone-delete.txt | cut -c1-400
