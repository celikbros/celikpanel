# configuration file /etc/nginx/conf.d/celikpanel-webmail.conf:
# Managed by CelikPanel — Roundcube webmail, Unix socket only.
# The panel reverse-proxies /webmail/ here; Roundcube's own login is the auth.
server {
    listen unix:/run/celikpanel-webmail.sock;
    server_name _;
    client_max_body_size 25m;

    location /webmail/ {
        alias /var/lib/celikpanel-webmail/public_html/;
        index index.php;
        try_files $uri $uri/ /webmail/index.php$is_args$args;

        location ~ ^/webmail/(.+\.php)$ {
            alias /var/lib/celikpanel-webmail/public_html/$1;
            include fastcgi_params;
            fastcgi_param SCRIPT_FILENAME $request_filename;
            fastcgi_pass unix:/run/php/php8.3-fpm.sock;
        }
    }
}

        fastcgi_param PATH_INFO $path_info;
        fastcgi_index index.php;
        include fastcgi.conf;
        fastcgi_pass unix:/var/run/php/php8.3-fpm-site2.sock;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        include fastcgi_params;

    location ~ /\.(?!well-known).* {
        deny all;
    }
}
